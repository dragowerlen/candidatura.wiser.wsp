const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbw0iCCax1KWiz-yW2QIg6tJSP9fwHzui2XGrmrsoS3aIPmpyjwyvfGvsOOT98-Tdp9Bg/exec"; const CHUNK_SIZE = 10 * 1024 * 1024; // 10 MB por parte async function submitForm(event) { event.preventDefault(); const fileInput = document.getElementById("video"); const file = fileInput.files[0]; if (!file) { alert("Selecione um vídeo."); return; } showLoading("Preparando envio..."); try { // 1. Cria sessão de upload no Drive const session = await fetch(APPS_SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "createSession", fileName: file.name, mimeType: file.type || "video/mp4", size: file.size }) }).then(r => r.json()); if (session.status !== "ok") { throw new Error("Não foi possível iniciar o upload: " + (session.reason || "")); } // 2. Envia o vídeo em partes const totalSize = file.size; let start = 0; while (start < totalSize) { const end = Math.min(start + CHUNK_SIZE - 1, totalSize - 1); const chunk = file.slice(start, end + 1); showLoading(`Enviando vídeo... ${Math.round((start / totalSize) * 100)}%`); const base64 = await blobToBase64(chunk); const result = await fetch(APPS_SCRIPT_URL, { method: "POST", body: JSON.stringify({ action: "uploadChunk", uploadUrl: session.uploadUrl, range: `bytes ${start}-${end}/${totalSize}`, chunkData: base64, totalSize: totalSize, fileName: file.name, name: document.getElementById("name").value.trim(), email: document.getElementById("email").value.trim(), phone: document.getElementById("phone").value.trim() }) }).then(r => r.json()); if (result.status === "ok") { showSuccess(result.videoLink); return; } if (result.status === "partial") { start = result.nextByte || end + 1; } else { throw new Error(result.reason || "Erro no upload da parte"); } } } catch (err) { console.error(err); showError("Falha no envio. Verifique sua internet e tente novamente."); } } function blobToBase64(blob) { return new Promise((resolve, reject) => { const reader = new FileReader(); reader.onload = () => resolve(reader.result.split(",")[1]); reader.onerror = reject; reader.readAsDataURL(blob); }); }
